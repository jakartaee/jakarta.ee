{{/* Renders a menu link with external link detection Usage: {{ partial "menu_link.html" (dict "url" .URL "name" .Name "context" $) }} */}} {{- $url := .url -}} {{- $name := .name -}} {{- $isExternal := false -}} {{- if hasPrefix $url "http://" -}} {{- $isExternal = true -}} {{- else if hasPrefix $url "https://" -}} {{- $isExternal = true -}} {{- else if hasPrefix $url "//" -}} {{- $isExternal = true -}} {{- end -}} {{- if not $isExternal -}} {{- with .page -}} {{/* Hugo resolves pageRef against the default language in a shared menu config, so .RelPermalink here is typically the English URL. We use it as the fallback, then check .Translations for the current rendering language. */}} {{- $currentLang := $.context.Lang -}} {{- $url = .RelPermalink -}} {{- range .Translations -}} {{- if eq .Lang $currentLang -}} {{- $url = .RelPermalink -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $name -}}
