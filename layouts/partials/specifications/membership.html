Membership comes from the platform diagram data file. Its columns are nested (Platform > Web Profile > Core Profile), so a spec listed in a column belongs to that profile and to every column before it. The profile keys are the folder names of the Platform and Profile specifications. Specs that aren't listed are standalone and have no entry here. Usage: {{ $membership := partialCached "specifications/membership.html" "jakartaee_12_platform_diagram" }} */}} {{ $diagram := index site.Data . }} {{ $profiles := slice "platform" "webprofile" "coreprofile" }} {{ $membership := dict }} {{ range $i, $column := $diagram.columns }} {{ $columnProfiles := first (add $i 1) $profiles }} {{ range $column.specs }} {{ $slug := index (split (strings.Trim .url "/") "/") 1 }} {{ $current := index $membership $slug | default slice }} {{ $membership = merge $membership (dict $slug (union $current $columnProfiles)) }} {{ end }} {{ end }} {{ return $membership }}
